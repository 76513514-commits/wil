import java.util.Scanner;

public class Cuadrados {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int N = sc.nextInt();

        for (int i = 1; i <= N; i++) {
            System.out.print(i * i);
            if (i < N) System.out.print(" ");
        }

        sc.close();
    }
}
